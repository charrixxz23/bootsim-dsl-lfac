== a. arranque completo correcto
   POWER_ON         ()                           ok -> ENCENDIDO
   CHECK_CPU        (4,)                         ok -> ENCENDIDO
   CHECK_RAM        (16,)                        ok -> ENCENDIDO
   CHECK_DISK       (512, 'GPT')                 ok -> ENCENDIDO
   RUN_POST         ()                           ok -> POST_OK
   SELECT_BOOT      ('DISK',)                    ok -> DISPOSITIVO_ELEGIDO
   LOAD_BOOTLOADER  ('GRUB',)                    ok -> GESTOR_CARGADO
   LOAD_KERNEL      ('linux-6.1', True)          ok -> KERNEL_CARGADO
   START_OS         ()                           ok -> SO_ACTIVO
== b. orden ilegal: kernel antes del post
   POWER_ON         ()                           ok -> ENCENDIDO
   LOAD_KERNEL      ('linux-6.1', True)          RECHAZADO LOAD_KERNEL: orden ilegal, estado ENCENDIDO y se necesitaba GESTOR_CARGADO
== c. post sin lecturas de hardware
   POWER_ON         ()                           ok -> ENCENDIDO
   RUN_POST         ()                           ERROR SIMULADO ERROR_POST: faltan lecturas de ['cpu', 'ram', 'disco']
== d. ram igual a cero (error de post)
   POWER_ON         ()                           ok -> ENCENDIDO
   CHECK_CPU        (2,)                         ok -> ENCENDIDO
   CHECK_RAM        (0,)                         ok -> ENCENDIDO
   CHECK_DISK       (64, 'GPT')                  ok -> ENCENDIDO
   RUN_POST         ()                           ERROR SIMULADO ERROR_POST: no se detecta memoria ram (en un equipo real suenan pitidos)
   SELECT_BOOT      ('DISK',)                    RECHAZADO SELECT_BOOT: el sistema esta en ERROR_POST; solo SHOW_STATUS o RESET_SYSTEM

== g. kernel sin initramfs y reinicio
   POWER_ON         ()                           ok -> ENCENDIDO

   POWER_ON         ()                           RECHAZADO POWER_ON: el sistema esta en KERNEL_PANIC; solo SHOW_STATUS o RESET_SYSTEM
