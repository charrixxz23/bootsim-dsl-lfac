## 3. un parser minimo de ply que solo importa tokens y crear_lexer (codigo de la demostracion y salida)
    # demostracion: un parser minimo de ply usa el lexer definitivo sin copiar nada (solo importa tokens y crear_lexer)
    import sys
    
    sys.path.insert(0, r'C:\Users\ADMIN003\Downloads\BootSimDSL\codigo')
    import ply.yacc as yacc
    
    from lexer import crear_lexer, tokens  # tokens lo necesita ply.yacc en este modulo
    
    
    def p_programa(p):
        """programa : lista"""
        p[0] = p[1]
    
    
    def p_lista_una(p):
        """lista : comando PUNTOYCOMA"""
        p[0] = [p[1]]
    
    
    def p_lista_mas(p):
        """lista : lista comando PUNTOYCOMA"""
        p[0] = p[1] + [p[2]]
    
    
    def p_comando(p):
        """comando : POWER_ON
                   | RUN_POST
                   | SHOW_STATUS"""
        p[0] = p[1]
    
    
    def p_error(p):
        print('error sintactico en', p)
    
    
    analizador = yacc.yacc(debug=False, write_tables=False)
    lexer = crear_lexer()
    print('resultado:', analizador.parse('POWER_ON; # arranque\nRUN_POST; SHOW_STATUS;', lexer=lexer))
    print('errores lexicos del lexer usado:', lexer.errores)
    lexer2 = crear_lexer()
    print('resultado con un @ :', analizador.parse('POWER_ON @;', lexer=lexer2))
    print('errores lexicos del segundo lexer:', [(e.linea, e.columna, e.fragmento) for e in lexer2.errores])
    print('errores lexicos del primero siguen igual:', lexer.errores)
